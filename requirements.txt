# 상한(<N)을 두는 이유: 워크플로우는 매 실행마다 pip이 인덱스에서 최신
# 버전을 새로 해석한다(setup-python의 cache: "pip"는 다운로드 캐시일 뿐
# 버전을 고정해주지 않는다). 상한이 없으면 우리가 코드를 한 줄도 안 고친
# 날에 상위 호환이 깨지는 메이저 버전이 올라와 그날 수집이 통째로 실패할
# 수 있고, 원인을 찾기도 어렵다. 하한은 보안 패치를 계속 받기 위해 열어둔다.

# 팀 로고에서 상단바 색을 추출 (generate_pages.py)
# 주의: Pillow 12에서도 이 코드가 쓰는 API(Image.open 컨텍스트 매니저,
# convert/resize/getdata)는 정상 동작하는 것을 확인했다. <12 상한을 유지할
# 특별한 이유가 없다면 <13으로 완화해도 된다.
Pillow>=10,<12

# 모든 외부 API 호출 (_common.fetch_json, subset_font 다운로드)
requests>=2.31,<3

# 구글 시트 읽기/쓰기 (_common.py)
# 이 프로젝트의 의존성 중 실제로 파괴적 변경 이력이 있는 유일한 패키지다.
# gspread 6.0에서 Worksheet.update()의 인자 순서가
# (range_name, values) -> (values, range_name)으로 뒤집혔다. 다행히 이
# 코드는 항상 키워드 인자(values=, range_name=)로 호출해서 5.12+/6.x 양쪽과
# 호환되지만, 다음 메이저에서 또 바뀌면 막아줄 장치가 지금은 없다.
gspread>=5.12,<7

# 서비스 계정 인증 (_common.get_gspread_client)
google-auth>=2.23,<3

# 폰트 서브셋 (subset_font.py에서 python -m fontTools.subset으로 호출)
# [woff] extra가 brotli>=1.0.1을 함께 설치하며, 이게 없으면 --flavor=woff2가
# 실패한다. 해당 워크플로우 스텝은 continue-on-error라 조용히 넘어가므로,
# 이 extra를 빠뜨리면 폰트가 갱신되지 않는 걸 한참 모를 수 있다.
fonttools[woff]>=4.44,<5

# 페이지 템플릿 렌더링 (generate_pages.py)
# 주의: 이 프로젝트는 autoescape를 끄고 쓴다(JS/CSS를 렌더하므로).
# <script> 안에 들어가는 값은 반드시 generate_pages.json_for_script()를 거칠 것.
jinja2>=3.1,<4
